# Symbols for MS Flight Simulator 1.05 (used by tools/fs1dis.py)
#
#   code SEG:OFF name                ; comment
#   var  SEG:OFF name [byte|word|dword|table|text] ; comment
#
# Code lives in segment 0050 (linear 00500-0685F). The game's data segment is DS=0618
# (stored by the loader at 0000:0120); it overlaps the tail of the code block.
# Fixed-point control values are signed 16-bit, 0x7FFF = full deflection.
# Evidence tags: [key] = changed by scripted key presses, [code] = read from disassembly.

# ---- boot / init -------------------------------------------------------------------------
code 0050:0000 boot_start            ; boot sector relocated from 0000:7C00
code 0050:0104 boot_read_track       ; int 13h read of track [03B3] into B800:0000 (also used for scenery)
code 0050:017C boot_stream_byte      ; next byte of the load stream
code 0050:01A0 boot_stream_word
code 0050:5600 cga_program_regs      ; writes (port-3D0, value) pairs from table [BX] (default 399C)
code 0050:5C9F start                 ; entry after the loader finishes
code 0050:5DA6 hook_int9             ; installs keyboard handler 0970, old vector saved at 03FB
code 0050:5DC3 hook_int8             ; installs timer handler 07A0, old vector saved at 040E

# ---- interrupt handlers ----------------------------------------------------------------
code 0050:07A0 int8_timer            ; IRQ0 handler
code 0050:0970 int9_keyboard         ; IRQ1 handler: port 60h, Ctrl+Alt+Del check, dispatch
code 0050:09AB key_dispatch          ; scancode -> handler via table key_handlers
